% pod-B serves player B's iframe. They share the same Cassandra +

\begin{tikzpicture}[x=1cm,y=1cm,
                    every node/.style={font=\sffamily\footnotesize}]
  \snapmarbleline{ctrl}{4.4}{POST /controls (lvl=1)}{0}{12.6}

  \draw[marbleline,red!70!black,dashed] (filter.east) -- (drop.west);
