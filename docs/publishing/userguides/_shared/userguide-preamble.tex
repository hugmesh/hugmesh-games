% Shared preamble for the six Snapdragon Games user guide PDFs.
%
% Each per-guide guide.tex \input's this file from the guide's working
% directory. Keeping it in one place means the architecture PDF and the
% six user guides share fonts, colours, code-listing styles, callout
% boxes, and the \snapdiagram / \snapmarble macros — they all look like
% one product.
%
% Path note: the snapdragon-theme.sty lives under
% docs/publishing/architecture/. We add that directory to TEXINPUTS via
% the build script (build-userguides.sh) so \usepackage finds it
% without copying the file into every guide directory.

\usepackage{datetime}
\usepackage{snapdragon-theme}
\usepackage{microtype}
\usepackage[hidelinks]{hyperref}
% pdfpages: lets the data-architect guide drop the auto-sized DB-diagram
% PDFs onto their own zoomable pages.
\usepackage{pdfpages}
% afterpage: queues per-page geometry hooks; the auto-generated
% schema-diagram includes use it to override \pdfpagewidth/\pdfpageheight
% per diagram (each diagram has its own natural page size).
\usepackage{afterpage}

\sloppy

% Override the chapter heading to read "PART" — user guides aren't
% chapters of one book, they're standalone parts under a per-guide title.
\renewcommand{\chaptermark}[1]{\markboth{#1}{}}

% Footer: project name + build timestamp on every page so each user
% guide carries the same provenance line as the architecture PDF.
% The theme already sets a footer; override here to point at the
% per-guide subtitle (\leftmark) instead of the fixed Architecture
% Reference label, while keeping the project-name + datetime block.
\fancyfoot[C]{\small\color{snapMuted}\snapbuildstamp}
